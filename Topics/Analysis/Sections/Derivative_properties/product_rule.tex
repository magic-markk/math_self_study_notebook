\subsubsection{Product rule}
For any two functions \(f\) and \(g\), we have that
\begin{equation}
    \left(f(x)g(x)\right)' = f(x)g'(x) + f'(x)g(x)
\end{equation}

\textbf{Proof}

\begin{align*}
    \frac{d}{dx}f(x)g(x) &= \lim_{a\to x} \frac{f(x)g(x) - f(a)g(a)}{x - a} \\
    &= \lim_{a \to x} \frac{f(x)g(x)}{x - a} - \frac{f(a)g(a)}{x - a} \\
    &= \lim
\end{align*}
