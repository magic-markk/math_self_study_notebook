\subsubsection{Sum rule}
Let \(f : A \to B\) be a function consisting of the sum of \(n\) functions, where each function is a term in the sequence \(\big(f_k(x)\big)_{k \le n} = (f_1(x), f_2(x),\dots, f_n(x))\):
\begin{equation}
    f(x) = \sum_{k=1}^{n}f_k(x)
\end{equation}

Sum rule:
\begin{equation}
    \left(\sum_{k=1}^{n}f_k(x)\right)' = \sum_{k=1}^{n}f_k'(x)
\end{equation}

Proof by induction:
\begin{equation}
    \statement(n) := \left(\sum_{k=1}^{n}f_k(x)\right)' = \sum_{k=1}^{n}f_k'(x)
\end{equation}

Base case\footnote{Base case is actually 1 for this proof to be valid, but that's trivial and boring so I wanna do 2 instead.} of \(\statement(n)\) for \(n=2\):
\begin{align*}
    f(x) &= f_1(x) + f_2(x) \\
    \implies f'(x) &= \lim_{h \to 0}\frac{f(x + h) - f(x)}{h} \\
    &= \lim_{h \to 0} \frac{(f_1(x + h) + f_2(x + h)) - (f_1(x) + f_2(x))}{h} \\
    &= \left(\lim_{h \to 0} \frac{f_1(x + h) - f_1(x)}{h}\right) + \left(\lim_{h \to 0} \frac{f_2(x + h) - f_2(x)}{h}\right) \\
    &= f_1'(x) + f_2'(x) & \text{By definition 1.2.5}
\end{align*}

Induction step:
\begin{align*}
    \Bigg(\Big(\statement(2) \land \statement(n)\Big) \implies \bigg(f(x) &= \sum_{k=1}^{n+1}f_k(x) \\
    &= f_{n+1}(x) + \sum_{k=1}^{n}f_k(x) \\
    \implies f'(x) &= \bigg(f_{n+1}(x) + \sum_{k=1}^{n}f_k(x)\bigg)' \\
    &= f_{n+1}'(x) + \bigg(\sum_{k=1}^{n}f_k(x)\bigg)' & \text{By } \statement(2) \\
    &= f_{n+1}'(x) + \sum_{k=1}^{n}f_k'(x) & \text{By } \statement(n) \\
    &= \sum_{k=1}^{n+1}f_k'(x)\bigg)\Bigg) \\
    &\implies \statement(n+1) & \blacksquare
\end{align*}
